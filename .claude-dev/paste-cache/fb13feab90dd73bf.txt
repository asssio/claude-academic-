import requests
from bs4 import BeautifulSoup
import re
import os

def fetch_blog_as_md(url, output_file='article.md'):
    """
    爬取博客园文章并保存为 Markdown 文件

    Args:
        url: 文章URL
        output_file: 输出的Markdown文件名
    """
    headers = {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36'
    }

    try:
        response = requests.get(url, headers=headers, timeout=15)
        response.raise_for_status()
        response.encoding = 'utf-8'
    except requests.exceptions.RequestException as e:
        print(f"❌ 网络请求失败: {e}")
        return

    soup = BeautifulSoup(response.text, 'html.parser')

    # ---- 提取标题 ----
    title_elem = soup.find('h1', class_='postTitle')
    title = title_elem.get_text(strip=True) if title_elem else "无标题"

    # ---- 提取作者和发布日期 ----
    post_info = soup.find('div', id='post_info')
    if post_info:
        info_text = post_info.get_text(strip=True)
        # 提取日期 (如 2024-03-15 14:30)
        date_match = re.search(r'(\d{4}-\d{2}-\d{2}\s+\d{2}:\d{2})', info_text)
        date = date_match.group(1) if date_match else "未知日期"
        # 提取作者（通常出现在 "posted @ 日期 作者" 或 "作者 发布于 ..."）
        # 尝试多种模式
        author_match = re.search(r'posted @.*?\s+([^\d]\S+)', info_text)
        if not author_match:
            author_match = re.search(r'作者[：:]\s*(\S+)', info_text)
        author = author_match.group(1) if author_match else "未知作者"
    else:
        date = "未知日期"
        author = "未知作者"

    # ---- 提取正文并转换为Markdown ----
    body_elem = soup.find('div', id='cnblogs_post_body')
    if not body_elem:
        print("❌ 未找到正文内容")
        return

    md_lines = []

    # 遍历所有直接子节点（包括文本节点和标签节点）
    for elem in body_elem.children:
        if elem.name is None:  # 文本节点（通常是换行或空白）
            continue

        if elem.name in ['h1', 'h2', 'h3', 'h4', 'h5', 'h6']:
            # 标题
            level = int(elem.name[1])
            text = elem.get_text(strip=True)
            if text:
                md_lines.append(f"{'#' * level} {text}\n")

        elif elem.name == 'p':
            # 段落
            text = elem.get_text(strip=True)
            if text:
                # 如果段落内包含代码块，将代码块用围栏包裹（但一般的段落内不会包含代码块）
                # 为了处理可能的内联代码，我们用 get_text 会把代码标签的文本也提取出来，但会丢失格式。
                # 简单处理：直接取文本
                md_lines.append(text + '\n')

        elif elem.name in ['pre', 'div']:
            # 代码块 - 查找内部的 code 标签，如果没有则取整个 pre 文本
            code_elem = elem.find('code')
            if code_elem:
                code_text = code_elem.get_text()
            else:
                code_text = elem.get_text()

            # 保留原始缩进，去除多余的首尾空行
            code_text = code_text.rstrip('\n')
            if code_text:
                md_lines.append('\n```\n' + code_text + '\n```\n')

        elif elem.name in ['ul', 'ol']:
            # 列表项（简单处理）
            items = elem.find_all('li', recursive=False)
            for li in items:
                text = li.get_text(strip=True)
                if text:
                    prefix = '- ' if elem.name == 'ul' else '1. '
                    md_lines.append(prefix + text)
            md_lines.append('')  # 空行分隔

        elif elem.name in ['blockquote']:
            # 引用块
            text = elem.get_text(strip=True)
            if text:
                md_lines.append(f'> {text}\n')

        elif elem.name == 'hr':
            md_lines.append('---\n')

        # 其他标签如 div, span 等忽略（它们的内容通常在子标签中已处理）

    # 合并正文，去除多余空行（保留最多一个空行）
    content = '\n'.join(md_lines)

    # ---- 组装完整Markdown ----
    md_content = f"# {title}\n\n"
    md_content += f"> **作者**：{author} ｜ **发布日期**：{date}\n\n"
    md_content += content

    # ---- 写入文件 ----
    try:
        with open(output_file, 'w', encoding='utf-8') as f:
            f.write(md_content)
        print(f"✅ 文章已保存至：{os.path.abspath(output_file)}")
    except IOError as e:
        print(f"❌ 保存文件失败: {e}")


if __name__ == '__main__':
    url = 'https://www.cnblogs.com/day1024/p/18948051'
    fetch_blog_as_md(url, 'article.md')
